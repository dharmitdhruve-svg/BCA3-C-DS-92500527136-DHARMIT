/* Unit 5 - Q2: Program to find out DFS
   Marwadi University - BCA | Language: C */

#include <stdio.h>

#define MAX 20

int adj[MAX][MAX];
int visited[MAX];
int n;

/* Go as deep as possible, then backtrack (recursion uses the stack) */
void dfs(int v) {
    int i;
    visited[v] = 1;
    printf("%d ", v);
    for (i = 0; i < n; i++) {
        if (adj[v][i] == 1 && !visited[i])
            dfs(i);
    }
}

int main() {
    int i, j, start;
    printf("Enter number of vertices (max %d): ", MAX);
    scanf("%d", &n);
    printf("Enter adjacency matrix (%d x %d), 1 = edge, 0 = no edge:\n", n, n);
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &adj[i][j]);
    for (i = 0; i < n; i++)
        visited[i] = 0;
    printf("Enter starting vertex (0 to %d): ", n - 1);
    scanf("%d", &start);
    printf("DFS Traversal: ");
    dfs(start);
    printf("\n");
    return 0;
}
